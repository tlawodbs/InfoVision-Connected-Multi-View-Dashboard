<script>
const svg = d3.select("svg"),
      margin = {top: 20, right: 150, bottom: 80, left: 60},
      width = parseInt(svg.style("width")) - margin.left - margin.right,
      height = parseInt(svg.style("height")) - margin.top - margin.bottom;

const g = svg.append("g").attr("transform", `translate(${margin.left},${margin.top})`);

const color = d3.scaleOrdinal(d3.schemeCategory10);
const tooltip = d3.select("body").append("div")
    .attr("class", "tooltip")
    .style("opacity", 0);

// Load and preprocess data
d3.csv("chocolate_sales.csv", d => ({
    product: d["Product"],
    date: d3.timeParse("%d-%b-%y")(d["Date"]),
    amount: +d["Amount"].replace(/[$,]/g, "")
})).then(data => {
    let products = Array.from(new Set(data.map(d => d.product)));
    const months = Array.from(new Set(data.map(d => d3.timeFormat("%Y-%m")(d.date)))).sort();

    const renderChart = (selectedProducts) => {
        const filteredProducts = products.filter(p => selectedProducts.includes(p));

        const monthlyData = months.map(month => {
            const monthData = {month};
            filteredProducts.forEach(prod => {
                monthData[prod] = d3.sum(data.filter(d => d3.timeFormat("%Y-%m")(d.date) === month && d.product === prod), d => d.amount);
            });
            return monthData;
        });

        const stack = d3.stack().keys(filteredProducts);
        const stackedData = stack(monthlyData);

        const x = d3.scaleBand().domain(months).range([0, width]).padding(0.1);
        const y = d3.scaleLinear().domain([0, d3.max(stackedData, d => d3.max(d, d => d[1]))]).nice().range([height, 0]);

        g.selectAll(".axis").remove();
        g.append("g").attr("class", "axis").attr("transform", `translate(0,${height})`)
          .call(d3.axisBottom(x).tickFormat(d => d3.timeFormat("%b %Y")(new Date(d))))
          .selectAll("text").attr("transform", "rotate(-45)").style("text-anchor", "end");

        g.append("g").attr("class", "axis").call(d3.axisLeft(y).ticks(6).tickFormat(d => `$${d3.format(",")(d)}`));

        const layer = g.selectAll(".layer").data(stackedData, d => d.key);

        layer.enter().append("g").attr("class", "layer").attr("fill", d => color(d.key))
            .selectAll("rect").data(d => d).join("rect")
            .attr("x", d => x(d.data.month)).attr("width", x.bandwidth())
            .attr("y", height).attr("height", 0)
            .on("mouseover", (event, d) => {
                const category = event.target.parentNode.__data__.key;
                tooltip.transition().duration(200).style("opacity", 0.9);
                tooltip.html(`<strong>${category}</strong><br>${d.data.month}<br>Sales: $${(d[1]-d[0]).toLocaleString()}`)
                    .style("left", (event.pageX + 10) + "px")
                    .style("top", (event.pageY - 30) + "px");
            }).on("mouseout", () => tooltip.transition().duration(200).style("opacity", 0))
            .transition().duration(800)
            .attr("y", d => y(d[1])).attr("height", d => y(d[0]) - y(d[1]));

        layer.exit().remove();
    };

    const checkboxContainer = d3.select(".checkboxes");
    products.forEach(prod => {
        const label = checkboxContainer.append("label");
        label.append("span")
            .attr("class", "color-box")
            .style("background-color", color(prod));
        label.append("input")
            .attr("type", "checkbox")
            .attr("checked", true)
            .attr("value", prod)
            .on("change", () => {
                const checked = d3.selectAll("input[type='checkbox']").nodes().filter(d => d.checked).map(d => d.value);
                renderChart(checked);
            });
        label.append("span").text(` ${prod}`);
    });

    renderChart(products);

    window.addEventListener("resize", () => {
        const newWidth = parseInt(svg.style("width")) - margin.left - margin.right;
        x.range([0, newWidth]);
        renderChart(products.filter(p => d3.select(`input[value='${p}']`).property("checked")));
    });
});
</script>
